import pandas as pd, numpy as np, re, os, json, math
# Clean baseline and records
def to_str(x):
    if pd.isna(x):
        return None
    s=str(x)
    if s.endswith('.0') and re.fullmatch(r'\d+\.0', s):
        s=s[:-2]
    return s

for col in b.columns:
    b[col]=b[col].apply(to_str)
    
# standardize section numeric string
def norm_section(x):
    if x is None or x=='' or str(x).lower()=='nan':
        return None
    s=to_str(x)
    s=s.replace('号','').strip()
    if re.fullmatch(r'\d+', s):
        return s
    return s
b['section_norm_std']=b['section_norm'].apply(norm_section)

# strat family
def strat_family(s):
    s=to_str(s)
    if not s:
        return None
    s=s.replace(' ', '')
    if s=='T3':
        return 'T3'
    if s.startswith('长'):
        m=re.match(r'长(\d+)', s)
        if m:
            digits=m.group(1)
            if len(digits)>=1:
                return '长'+digits[0]
    return s

b['strat_family']=b['strat_unit_norm'].apply(strat_family)
