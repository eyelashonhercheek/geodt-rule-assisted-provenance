# Helper
def col_letter(n):
    s=''
    while n:
        n,r=divmod(n-1,26)
        s=chr(65+r)+s
    return s

def write_sheet(wb, name, rows):
    sh=wb.worksheets.add(name)
    if not rows:
        return sh
    nrows=len(rows); ncols=max(len(r) for r in rows)
    # normalize length
    vals=[list(r)+[""]*(ncols-len(r)) for r in rows]
    # write in chunks of 500 rows to be safe
    start=1
    for i in range(0,nrows,500):
        chunk=vals[i:i+500]
        rng=f"A{start}:{col_letter(ncols)}{start+len(chunk)-1}"
        sh.get_range(rng).values=chunk
        start+=len(chunk)
    # format header
    sh.get_range(f"A1:{col_letter(ncols)}1").format.font.bold=True
    sh.get_range(f"A1:{col_letter(ncols)}1").format.fill.color="#E8F1FB"
    # auto filter? maybe help? skip
    # set widths for columns based on name
    try:
        sh.get_range(f"A1:{col_letter(ncols)}{min(nrows,200)}").format.autofit_columns()
    except Exception as e:
        pass
    return sh
